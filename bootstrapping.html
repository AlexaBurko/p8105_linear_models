Simulated regression

n_samp = 250

sim_df_const =
  tibble(
    x = rnorm(n_samp, 1, 1),
    error = rnorm(n_samp, 0, 1),
    y = 2 + 3 * x + error
  )

sim_df_nonconst =
  sim_df_const |> 
  mutate(
    error = error * .75 * x,
    y = 2 + 3 * x + error
  )

draw_boot_sample = function(df) {
 
   sample_frac(df, replace = TRUE)
  
  
}

Draw a bunch of samples and analyze

boot_strap_df =
  tibble(
    strap_number = 1:1000
  ) |> 
  mutate(
    strap_sample = map(strap_number, \(i) draw_boot_sample(sim_df_nonconst)),
    model_fit = map(strap_sample, \(df) lm(y~x, data = df)),
    results = map(model_fit, broom::tidy)
  ) |> 
  unnest(results)

did this work

boot_strap_df |> 
  group_by(term) |> 
  summarise(
    mean_est = mean(estimate),
    se_est = sd(estimate)
